## The same policy, twice

```yaml title="governance.yml"
orgs:
  my-org:
    repos:
      api:
        hasWiki: false
        deleteBranchOnMerge: true
        topics: [service, api]
        branchProtection:
          - pattern: main
            requirePullRequestReviews: true
            requiredApprovingReviewCount: 1
```

```ts title="governance.ts"
import type { GovernanceConfig } from "@intentius/forgejo-warden";

const protectedMain = { pattern: "main", requirePullRequestReviews: true, requiredApprovingReviewCount: 1 };

export default {
  orgs: {
    "my-org": {
      repos: {
        api: { hasWiki: false, deleteBranchOnMerge: true, topics: ["service", "api"], branchProtection: [protectedMain] },
      },
    },
  },
} satisfies GovernanceConfig;
```

The second one has completion, a type error on a misspelt key, and a `const` it can reuse across forty repositories. The tool that reads it gets the same object either way, and it never runs the file to get it.
